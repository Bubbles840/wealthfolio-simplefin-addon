# Platform upgrades (1.62.0 Part 3, U1–U4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The add-on builds against SDK 3.9.0 and makes each 3.9 call only where the host offers it (U1); both syncers and both Links link transfers in place through Wealthfolio's own link, keeping ids and filings, and Undo unlinks in place (U2); the app decides "linked" from Wealthfolio's own facts and its stories name a row's category and the other half of its transfer (U3); on Wealthfolio 3.10 a card's bookkeeping rows are TRANSFER_OUT, and the companion rewrites the existing ones once (U4).

**Architecture:** One feature check, `hostCan` (src/utils/host-can.ts), gates every SDK 3.9 call. Two pure shared modules carry the logic: `shared/link-in-place.ts` decides from the two stored rows whether Wealthfolio's own link gives exactly what re-creating gives, links or unlinks in place, and falls back to the existing `linkPairByRecreate` / `unlinkPairByRecreate` on every other shape and on any refusal; `shared/wf-facts.ts` reads a `CashActivity` (link status, group, bucket, filings). Each host lends its calls: the companion through `POST /activities/link` and `/unlink` (`RestSyncHost`), the add-on through the SDK (`AddonSyncHost`, `src/utils/wf-facts.ts`). The sync, R2's Link, Remove for good, Undo and the app's story consume them. U4 threads one capability, `cardTransferOut`, from `GET /app/info` (the companion) or the companion's published `wealthfolio_version` (the add-on) into the bookkeeping shapes, makes the two shapes one row to the planner, and adds a one-time, resumable rewrite in the companion.

**Tech Stack:** TypeScript, React (add-on), vitest (root project: jsdom, `src/` + `shared/`; companion project: node, `companion/src/`), Wealthfolio REST 3.8–3.10 (companion), add-on SDK 3.9.0 on Wealthfolio 3.8–3.10 (add-on), `node:sqlite` (the companion's read-only database reads).

```ts
/**
 * The add-on sandbox's `ctx.api` as Wealthfolio builds it (upstream
 * addon-sandbox-entry.tsx `createApiProxy`): every path is a function, and a
 * call the host does not list is refused with "Unknown addon host API method".
 */
function sandbox(allowed: readonly string[], impl: Record<string, (...args: unknown[]) => unknown> = {}) {
  const calls: string[] = [];
  const at = (path: string[]): unknown => new Proxy(function addonApiProxy() {}, {
    get: (_t, key) => (key === 'then' || typeof key === 'symbol' ? undefined : at([...path, String(key)])),
    apply: (_t, _this, args: unknown[]) => {
      const method = path.join('.');
      calls.push(method);
      if (!allowed.includes(method)) return Promise.reject(new Error(`Unknown addon host API method '${method}'`));
      try {
        return Promise.resolve(impl[method]?.(...args));
      } catch (err) {
        return Promise.reject(err);
      }
    },
  });
  return { ctx: { api: at([]) } as any, calls };
}
```

```ts
/**
 * src/utils/host-can.ts
 *
 * The one feature check for the add-on SDK 3.9 calls (1.62.0, U1). Every 3.9
 * call the add-on makes goes through `hostCan` first; where it says no, the
 * add-on does what it did on 3.8. `minWealthfolioVersion` stays 3.8.0.
 *
 * Why not `typeof ctx.api.activities?.linkTransfer === 'function'`: inside
 * Wealthfolio's add-on sandbox `ctx.api` is a Proxy on which EVERY path is a
 * callable function (upstream addon-sandbox-entry.tsx `createApiProxy`, 3.8
 * through 3.10). A 3.8 host only refuses the call: it rejects with "Unknown
 * addon host API method '<name>'" (addon-iframe-manager.ts `getMethod`,
 * against `ALLOWED_API_METHODS`). Nor can the probe be `spending.isEnabled`:
 * 3.8 already offers it, `getCategories` and the rule calls (type-bridge.ts,
 * v3.8.0). So the check makes the cheapest call only 3.9+ offers — a cash
 * search no row can match — once per context:
 *  - it resolves: a 3.9+ host, whose allowed-method list carries every call
 *    below (3.9.0 through 3.10.0);
 *  - "Unknown addon host API method" (a 3.8 host) or "is not allowed to call"
 *    (`AddonPermissionDenied`, a manifest without the permission): no, and
 *    remembered;
 *  - anything else (a passing failure): no for now, and not remembered.
 * A host object that has no such function at all (a test double, a future
 * host) is a plain no. Nothing else is ever called here.
 */
import type { AddonContext, CashActivitySearchRequest } from '@wealthfolio/addon-sdk';

/** The SDK 3.9 calls the add-on makes, by `<api namespace>.<method>`. */
export type Sdk39Method =
  | 'activities.linkTransfer'
  | 'activities.unlinkTransfer'
  | 'activities.getTransferPair'
  | 'spending.searchCashActivities';

/**
 * The manifest permission each call needs: `[category, function]`. Not always
 * its namespace — Wealthfolio guards `spending.searchCashActivities` under the
 * high-risk `activities` category (upstream type-bridge.ts, v3.9.0).
 */
export const SDK39_PERMISSIONS: Readonly<Record<Sdk39Method, readonly [string, string]>> = {
  'activities.linkTransfer': ['activities', 'linkTransfer'],
  'activities.unlinkTransfer': ['activities', 'unlinkTransfer'],
  'activities.getTransferPair': ['activities', 'getTransferPair'],
  'spending.searchCashActivities': ['activities', 'searchCashActivities'],
};

/** A cash search no row can match: one day in 1970, one row at most. */
const PROBE: CashActivitySearchRequest = {
  startDate: '1970-01-01T00:00:00.000Z', endDate: '1970-01-01T00:00:00.000Z', offset: 0, limit: 1,
};

/** The host's answer, per context. */
const answers = new WeakMap<object, Promise<boolean>>();

/** A refusal that will not change while the add-on runs. */
const LASTING = /Unknown addon host API method|is not allowed to call|AddonPermissionDenied/i;

const fnAt = (ctx: AddonContext, method: string): unknown => {
  const [ns, name] = method.split('.');
  const api = (ctx as unknown as { api?: Record<string, Record<string, unknown> | undefined> }).api;
  return api?.[ns]?.[name];
};

async function probe(ctx: AddonContext): Promise<{ yes: boolean; lasting: boolean }> {
  if (typeof fnAt(ctx, 'spending.searchCashActivities') !== 'function') return { yes: false, lasting: true };
  try {
    // Called as a method, never through `.call`: on the sandbox's Proxy
    // `.call` is one more path ("spending.searchCashActivities.call"), which
    // no host offers.
    await ctx.api.spending.searchCashActivities(PROBE);
    return { yes: true, lasting: true };
  } catch (err) {
    const text = err instanceof Error ? `${err.name} ${err.message}` : String(err);
    return { yes: false, lasting: LASTING.test(text) };
  }
}

/** Whether the host offers this SDK 3.9 call. Never rejects. */
export async function hostCan(ctx: AddonContext, method: Sdk39Method): Promise<boolean> {
  if (typeof fnAt(ctx, method) !== 'function') return false;
  const known = answers.get(ctx);
  if (known) return known;
  const asking = probe(ctx).then(({ yes, lasting }) => {
    if (!lasting) answers.delete(ctx);
    return yes;
  });
  answers.set(ctx, asking);
  return asking;
}
```

### Task 2: Link in place — the pure linker and unlinker

Spec U2's decision, from verified fact (b). `linkPairPreferInPlace` reads both rows by id, and links them through Wealthfolio's own link only when each stored row already is the leg re-creating would write (`inPlaceRefusal`: the type, the UTC day, the amount, the currency, the note, no asset, subtype or fee, and not marked external); everything else, a failed read, a refusal — and a link whose answer was lost and whose re-read shows no shared group — goes to `linkPairByRecreate`, as before. `unlinkPairPreferInPlace` unlinks through Wealthfolio, then retypes both rows to ordinary money in ONE bulk update (the stored instant kept; a non-transfer type carries the `$CASH-<ccy>` symbol), else `unlinkPairByRecreate`. `ledgerGroupId` keeps a Wealthfolio-minted group under the prefix the add-on's ledger keeps. `LinkResult` and `UnlinkResult` gain `inPlace`. /**
```ts
 * Wealthfolio in memory: the exact-id read, its link and unlink (one
 * transaction each; a refusal writes nothing), and the bulk endpoint.
 */
function wealthfolio(rows: Row[], opts: { refuseLink?: string; loseLinkAnswer?: boolean; refuseUnlink?: string; refuseUpdate?: string } = {}) {
  let n = 0;
  const calls: string[] = [];
  const asSearch = (r: Row) => ({
    id: r.id, accountId: r.accountId, activityType: r.activityType, date: r.date, amount: r.amount, currency: r.currency, comment: r.comment,
    assetId: r.assetId ?? '', subtype: r.subtype ?? null, fee: r.fee ?? '0', sourceGroupId: r.sourceGroupId, metadata: r.metadata ? JSON.parse(r.metadata) : null,
  });
  const flow = (r: Row, external: boolean) => JSON.stringify({ ...(r.metadata ? JSON.parse(r.metadata) : {}), flow: { is_external: external } });
  const io: InPlaceIo = {
    read: async (ids) => {
      calls.push('read');
      return rows.filter((r) => ids.includes(r.id)).map(asSearch);
    },
    link: async (a, b) => {
      calls.push(`link ${a} ${b}`);
      if (opts.refuseLink) throw new Error(opts.refuseLink);
      const pair = [a, b].map((id) => rows.find((r) => r.id === id)!);
      const group = `uuid-${++n}`;
      for (const r of pair) {
        r.sourceGroupId = group;
        r.metadata = flow(r, false);
      }
      if (opts.loseLinkAnswer) throw new Error('fetch failed');
      // Transfer-in first, as upstream answers.
      return [...pair].sort((x, y) => Number(y.activityType === 'TRANSFER_IN') - Number(x.activityType === 'TRANSFER_IN'))
        .map((r) => ({ id: r.id, accountId: r.accountId, activityType: r.activityType, activityDate: r.date, notes: r.comment, sourceGroupId: r.sourceGroupId }));
    },
    unlink: async (a, b) => {
      calls.push(`unlink ${a} ${b}`);
      if (opts.refuseUnlink) throw new Error(opts.refuseUnlink);
      const pair = [a, b].map((id) => rows.find((r) => r.id === id)!);
      for (const r of pair) {
        r.sourceGroupId = null;
        r.metadata = flow(r, true);
      }
      return pair.map(asSearch);
    },
  };
  const write = (w: ActivityWrite, row: Row) => Object.assign(row, {
    accountId: w.accountId, activityType: w.activityType, date: w.activityDate, amount: String(w.amount), currency: w.currency, comment: w.comment,
    ...(w.sourceGroupId !== undefined ? { sourceGroupId: w.sourceGroupId } : {}),
    ...(w.metadata !== undefined ? { metadata: w.metadata } : {}),
    ...(w.fee !== undefined ? { fee: String(w.fee) } : {}),
  });
  const saveMany = async (req: SaveManyRequest): Promise<SaveManyResult> => {
    calls.push(`saveMany${req.deleteIds?.length ? ' delete' : ''}${req.creates?.length ? ' create' : ''}${req.updates?.length ? ' update' : ''}`);
    if (req.updates?.length && opts.refuseUpdate) return { created: [], updated: [], errors: [{ action: 'update', message: opts.refuseUpdate }] };
    for (const id of req.deleteIds ?? []) rows.splice(rows.findIndex((r) => r.id === id), 1);
    const created = (req.creates ?? []).map((w) => {
      const row = write(w, { id: `new-${++n}`, sourceGroupId: null, metadata: null } as Row);
      rows.push(row);
      return { id: row.id, accountId: row.accountId, activityType: row.activityType, date: row.date, amount: row.amount, comment: row.comment, sourceGroupId: row.sourceGroupId };
    });
    const updated = (req.updates ?? []).map((w) => {
      const row = write(w, rows.find((r) => r.id === w.id)!);
      return { id: row.id, accountId: row.accountId, activityType: row.activityType, date: row.date, amount: row.amount, comment: row.comment, sourceGroupId: row.sourceGroupId };
    });
    return { created, updated, errors: [] };
  };
  return { io, saveMany, rows, calls };
}
```

```ts
/**
 * shared/link-in-place.ts
 *
 * Link and unlink two legs IN PLACE, through Wealthfolio's own link — `POST
 * /activities/link` and `/activities/unlink` (the companion, Wealthfolio 3.8+)
 * or the SDK's `activities.linkTransfer` and `unlinkTransfer` (the add-on, a
 * 3.9+ host). Ids stay stable, and so does everything filed on the rows.
 * Where in place cannot give the result re-creating gives, the leg shape goes
 * to the delete-and-re-create linker as before (shared/link-pair.ts). Pure:
 * each host lends its reads and calls (`InPlaceIo`).
 *
 * What upstream's link does (storage-sqlite activities repository.rs
 * `link_transfer_activities`, v3.8.0 through v3.10.0):
 *  - one transaction: both rows are read, checked and updated together, and a
 *    refusal writes nothing;
 *  - it needs one row whose effective type is TRANSFER_IN and one TRANSFER_OUT
 *    — it never changes a type — in two accounts (one account only for a cash
 *    FX conversion);
 *  - it refuses a row already in another VALID pair ("already linked to
 *    another transfer"); a row in an orphan or broken group is re-grouped;
 *    3.10 also accepts the pair already linked to each other (keeps its group);
 *  - it checks the asset shape only: two cash legs (no asset, or a `$CASH-XXX`
 *    one) pass; a security asset must match on both legs with a quantity. It
 *    does NOT clear an asset — so a leg carrying the phantom "$CASH" security
 *    (a uuid asset that `is_cash_symbol` rejects) is refused or, worse, linked
 *    and still books no cash;
 *  - it never checks amount, currency or date;
 *  - it writes a new uuid `source_group_id` (no `wf-transfer-` prefix),
 *    merges `flow.is_external: false` into each row's metadata, and sets
 *    `is_user_modified` (read only by provider upserts, which never touch
 *    these rows) — nothing else: type, date, amount, note, asset, subtype,
 *    fee and every category assignment stay as they are;
 *  - it answers with both rows, transfer-in first.
 * Unlink is its mirror: both rows must share one group; it clears the group,
 * sets `flow.is_external: true`, and changes nothing else.
 *
 * So in place gives exactly what re-creating gives only when each stored row
 * already IS the leg re-creating would write: the type, the UTC day, the
 * amount, the currency, the note, and no asset, subtype or fee. Re-creating
 * also drops a filing; in place keeps it — the point of linking in place —
 * except on a row marked external: the sync's held placeholder, whose
 * bookkeeping filing must not ride onto a real transfer, keeps re-creating.
 * Rows are read by id right before the call. Wealthfolio decides the group
 * question itself, atomically: a refusal (of any kind) falls back to
 * re-creating, which is how every pair linked before 1.62.0. A call whose
 * answer is lost is read again: a shared group means it went through.
 *
 * Unlinking in place keeps the rows and their ids, then retypes both to
 * ordinary money in ONE bulk update — they are no longer a pair, so an update
 * to one is not copied onto the other (Wealthfolio 3.8 `build_counterpart_update`
 * only follows a valid pair). If that update fails the pair stays unlinked,
 * as transfers marked external, and says so.
 */
import { describeLeg, linkPairByRecreate, unlinkPairByRecreate, type UnlinkLeg, type UnlinkResult } from './link-pair.js';
import { flowExternalOf, isTransferType, TRANSFER_GROUP_PREFIX } from './sync-core.js';
import type { ActivityWrite, LinkLeg, LinkResult, SaveManyRequest, SaveManyResult } from './sync-host.js';

/** What a host lends: Wealthfolio's link and unlink, and a read by id. */
export interface InPlaceIo {
  /** Both rows by id as the host's search returns them (a missing row is absent). Rejects when unreachable. */
  read(ids: [string, string]): Promise<unknown[]>;
  /** Wealthfolio's link: resolves to the two rows it wrote; rejects on a refusal. */
  link(aId: string, bId: string): Promise<unknown[]>;
  /** Wealthfolio's unlink: resolves to the two rows it wrote; rejects on a refusal. */
  unlink(aId: string, bId: string): Promise<unknown[]>;
}

/** A stored row as in-place linking checks it: a search row (ActivityDetails) or a link answer (Activity). */
export interface StoredLeg {
  id: string;
  accountId: string;
  /** Upper-case. */
  type: string;
  /** The stored instant, as the host sent it. */
  instant: string;
  /** YYYY-MM-DD: the instant's UTC day. */
  day: string;
  absCents: number;
  feeCents: number;
  /** Upper-case. */
  currency: string;
  notes: string;
  /** '' when none. */
  assetId: string;
  /** '' when none. */
  subtype: string;
  /** '' when none. */
  group: string;
  /** The host reported the group (sent the key, even empty). */
  groupKnown: boolean;
  /** `flow.is_external`: true/false; null when unset; undefined when unknown. */
  external: boolean | null | undefined;
}

const isRecord = (v: unknown): v is Record<string, unknown> => !!v && typeof v === 'object' && !Array.isArray(v);
const text = (v: unknown) => (typeof v === 'string' ? v.trim() : '');
const cents = (v: unknown) => Math.round(Math.abs(Number(v ?? 0)) * 100) || 0;
const dayOf = (v: string) => {
  const d = new Date(v);
  return Number.isNaN(d.getTime()) ? v.slice(0, 10) : d.toISOString().slice(0, 10);
};

/** A host row as in-place linking reads it; null without an id or an account. */
export function storedLegOf(raw: unknown): StoredLeg | null {
  if (!isRecord(raw)) return null;
  const id = text(raw.id);
  const accountId = text(raw.accountId);
  if (!id || !accountId) return null;
  const instant = typeof raw.date === 'string' ? raw.date : typeof raw.activityDate === 'string' ? raw.activityDate : '';
  return {
    id,
    accountId,
    type: text(raw.activityType).toUpperCase(),
    instant,
    day: dayOf(instant),
    absCents: cents(raw.amount),
    feeCents: cents(raw.fee),
    currency: text(raw.currency).toUpperCase(),
    notes: typeof raw.comment === 'string' ? raw.comment : typeof raw.notes === 'string' ? raw.notes : '',
    assetId: text(raw.assetId),
    subtype: text(raw.subtype),
    group: text(raw.sourceGroupId),
    groupKnown: 'sourceGroupId' in raw,
    external: 'metadata' in raw ? flowExternalOf(raw.metadata) : undefined,
  };
}

const find = (rows: unknown[], id: string): StoredLeg | null =>
  rows.map(storedLegOf).find((s): s is StoredLeg => s !== null && s.id === id) ?? null;

/** Why this pair cannot be linked in place (for a log), or null when it can. */
export function inPlaceRefusal(legs: readonly [LinkLeg, LinkLeg], stored: readonly [StoredLeg | null, StoredLeg | null]): string | null {
  const types = new Set(legs.map((l) => l.activityType));
  if (!types.has('TRANSFER_OUT') || !types.has('TRANSFER_IN')) return 'Wealthfolio links one transfer out with one transfer in';
  if (legs[0].accountId === legs[1].accountId) return 'both legs are in one account';
  for (let i = 0; i < 2; i++) {
    const leg = legs[i];
    const s = stored[i];
    if (!s) return `${leg.wfId} could not be read`;
    if (s.accountId !== leg.accountId) return `${leg.wfId} is in another account now`;
    if (s.type !== leg.activityType) return `${leg.wfId} is stored as ${s.type} and links as ${leg.activityType}`;
    if (s.day !== leg.date) return `${leg.wfId} is dated ${s.day}, not ${leg.date}`;
    if (s.absCents !== leg.absCents) return `${leg.wfId} has another amount`;
    if (s.currency !== leg.currency.toUpperCase()) return `${leg.wfId} is in ${s.currency}`;
    if (s.notes !== leg.comment) return `${leg.wfId} has another note`;
    if (s.assetId) return `${leg.wfId} carries an asset`;
    if (s.subtype) return `${leg.wfId} carries a subtype`;
    if (s.feeCents) return `${leg.wfId} carries a fee`;
    if (s.external === true) return `${leg.wfId} is marked as an outside transfer`;
  }
  return null;
}

/** The pair, linked: both rows share a group. Null when they do not. */
function linkedIn(legs: readonly [LinkLeg, LinkLeg], rows: unknown[]): LinkResult | null {
  const a = find(rows, legs[0].wfId);
  const b = find(rows, legs[1].wfId);
  if (!a || !b || !a.group || a.group !== b.group) return null;
  return { linked: true, groupId: a.group, activityIds: [legs[0].wfId, legs[1].wfId], inPlace: true };
}

/**
 * Links the pair in place where the stored rows allow it, and by
 * re-creating them everywhere else — `io` null (a host without Wealthfolio's
 * link), a read that fails, a shape in place would not keep, or a link
 * Wealthfolio refuses.
 */
export async function linkPairPreferInPlace(
  io: InPlaceIo | null,
  saveMany: (req: SaveManyRequest) => Promise<SaveManyResult>,
  legs: [LinkLeg, LinkLeg],
): Promise<LinkResult> {
  if (!io) return linkPairByRecreate(saveMany, legs);
  const ids: [string, string] = [legs[0].wfId, legs[1].wfId];
  let before: unknown[];
  try {
    before = await io.read(ids);
  } catch {
    return linkPairByRecreate(saveMany, legs);
  }
  if (inPlaceRefusal(legs, [find(before, ids[0]), find(before, ids[1])]) !== null) return linkPairByRecreate(saveMany, legs);
  try {
    const done = linkedIn(legs, await io.link(ids[0], ids[1]));
    if (done) return done;
  } catch {
    // Refused (nothing was written), or its answer was lost: look.
  }
  try {
    const done = linkedIn(legs, await io.read(ids));
    if (done) return done;
  } catch {
    // Can't tell: re-creating links them either way.
  }
  return linkPairByRecreate(saveMany, legs);
}

/** Why this pair cannot be unlinked in place, or null when it can. */
export function unlinkInPlaceRefusal(legs: readonly [UnlinkLeg, UnlinkLeg], stored: readonly [StoredLeg | null, StoredLeg | null]): string | null {
  const types = new Set(stored.map((s) => s?.type));
  if (!types.has('TRANSFER_OUT') || !types.has('TRANSFER_IN')) return 'Wealthfolio unlinks one transfer out from one transfer in';
  for (let i = 0; i < 2; i++) {
    const leg = legs[i];
    const s = stored[i];
    if (!s) return `${leg.wfId} could not be read`;
    if (s.accountId !== leg.accountId) return `${leg.wfId} is in another account now`;
    if (s.type !== leg.activityType) return `${leg.wfId} is stored as ${s.type}`;
    if (s.absCents !== leg.absCents) return `${leg.wfId} has another amount`;
    if (s.assetId) return `${leg.wfId} carries an asset`;
    if (s.groupKnown && !s.group) return `${leg.wfId} is not linked`;
  }
  const [a, b] = stored as [StoredLeg, StoredLeg];
  if (a.groupKnown && b.groupKnown && a.group !== b.group) return 'the two rows are in different groups';
  return null;
}

/** The leg as ordinary money, written over its own row: the stored instant
 *  kept, the cash symbol for any non-transfer type (see `isTransferType`). */
function retypeWrite(leg: UnlinkLeg, s: StoredLeg): ActivityWrite {
  return {
    id: leg.wfId,
    accountId: leg.accountId,
    activityType: leg.newType,
    activityDate: s.instant || leg.date,
    ...(isTransferType(leg.newType) ? {} : { symbol: { symbol: `$CASH-${leg.currency}` } }),
    amount: leg.absCents / 100,
    fee: 0,
    currency: leg.currency,
    comment: leg.comment,
  };
}

/**
 * Undo of a link, in place where it can be: Wealthfolio's unlink, then one
 * bulk update retyping both rows as ordinary money in or out. Everything else
 * — `io` null, a failed read, a shape it cannot take, a refused unlink —
 * goes to `unlinkPairByRecreate` as before.
 */
export async function unlinkPairPreferInPlace(
  io: InPlaceIo | null,
  saveMany: (req: SaveManyRequest) => Promise<SaveManyResult>,
  legs: [UnlinkLeg, UnlinkLeg],
): Promise<UnlinkResult> {
  if (!io) return unlinkPairByRecreate(saveMany, legs);
  const ids: [string, string] = [legs[0].wfId, legs[1].wfId];
  let before: unknown[];
  try {
    before = await io.read(ids);
  } catch {
    return unlinkPairByRecreate(saveMany, legs);
  }
  const stored: [StoredLeg | null, StoredLeg | null] = [find(before, ids[0]), find(before, ids[1])];
  if (unlinkInPlaceRefusal(legs, stored) !== null) return unlinkPairByRecreate(saveMany, legs);
  let unlinked = false;
  try {
    await io.unlink(ids[0], ids[1]);
    unlinked = true;
  } catch {
    // Refused, or its answer was lost: unlinked only if both rows now say so.
    try {
      const again = await io.read(ids);
      unlinked = ids.every((id) => {
        const s = find(again, id);
        return !!s && s.groupKnown && !s.group;
      });
    } catch {
      unlinked = false;
    }
  }
  if (!unlinked) return unlinkPairByRecreate(saveMany, legs);
  let problem: string | null = null;
  try {
    const res = await saveMany({ updates: legs.map((leg, i) => retypeWrite(leg, stored[i]!)) });
    if (res.errors.length > 0) problem = res.errors.map((e) => `${e.action}: ${e.message}`).join('; ');
  } catch (e: any) {
    problem = String(e?.message ?? e);
  }
  if (problem !== null) {
    return {
      unlinked: true, inPlace: true, activityIds: ids,
      problems: [`unlinked ${describeLeg(legs[0])} and ${describeLeg(legs[1])} in place, but they are still typed as transfers (${problem}); a later sync retypes each while the bank feed still sends it`],
    };
  }
  return { unlinked: true, inPlace: true, activityIds: ids };
}

/**
 * The add-on's `linked_groups` value for a pair. Its sync drops every entry
 * without the `wf-transfer-` prefix as written before the ledger was
 * echo-confirmed, and Wealthfolio's own link mints a bare uuid — so a pair
 * linked in place is remembered under the prefix. The value is only ever
 * compared between the two legs, never sent to Wealthfolio.
 */
export const ledgerGroupId = (groupId: string): string =>
  groupId.startsWith(TRANSFER_GROUP_PREFIX) ? groupId : `${TRANSFER_GROUP_PREFIX}${groupId}`;
```

```ts
/**
 * shared/wf-facts.ts
 *
 * What Wealthfolio itself says about a cash row (1.62.0, U3), read through
 * the add-on SDK's `spending.searchCashActivities` on a 3.9+ host: whether a
 * transfer leg is linked, whether the row carries a group, its cash-flow
 * bucket and what it is filed under. These are facts, so where one is read
 * the app stops guessing "maybe linked" (change-undo-run.ts
 * `maybeLinkedTransfer`, fix-row.ts `linkedState`). Pure.
 *
 * Upstream (crates/spending cash_activities/service.rs, v3.9.0):
 *  - `transferLinkStatus` is set for an effective TRANSFER_IN/OUT only:
 *    `linked` — half of a valid pair; `invalid` — it carries a group that is
 *    not a valid pair; `unlinked` — no group. Any other type has none.
 *  - The call returns rows of accounts opted in to Spending, of the cash
 *    types, that Wealthfolio shows: an unlinked transfer out of a card
 *    (Ignored, 3.10) is not among them. A row it does not return has NO fact,
 *    and the caller keeps its old rule for it.
 *  - It has no id filter: rows are found by account and an inclusive RFC 3339
 *    window, at most 1,000 a page (`offset`, `limit`, `totalCount`).
 * A row in any group — linked, broken, or a non-transfer someone grouped —
 * counts as linked: deleting it deletes the rest of its group.
 */
import { SPENDING_TAXONOMY_ID } from './cash-flow-bucket.js';

export type TransferLinkStatus = 'linked' | 'unlinked' | 'invalid';
export type FactBucket = 'spending' | 'income' | 'saving' | 'neutral';

export interface FactFiling {
  taxonomyId: string;
  categoryId: string;
  /** `manual`, `rule`, … — who Wealthfolio says filed it. */
  source: string;
}

export interface CashFact {
  activityId: string;
  accountId: string;
  /** Wealthfolio's link status for a transfer leg; null for any other type. */
  link: TransferLinkStatus | null;
  /** The row carries a transfer group id. */
  grouped: boolean;
  bucket: FactBucket | null;
  filings: FactFiling[];
}

/** The most rows one `searchCashActivities` page returns (upstream `MAX_CASH_ACTIVITY_SEARCH_LIMIT`). */
export const CASH_FACTS_PAGE = 1000;

const isRecord = (v: unknown): v is Record<string, unknown> => !!v && typeof v === 'object' && !Array.isArray(v);
const LINK = new Set<string>(['linked', 'unlinked', 'invalid']);
const BUCKET = new Set<string>(['spending', 'income', 'saving', 'neutral']);

/** One `CashActivity` as the app reads it; null when it is not one this build can read. */
export function cashFactOf(raw: unknown): CashFact | null {
  if (!isRecord(raw) || typeof raw.id !== 'string' || !raw.id || typeof raw.accountId !== 'string') return null;
  const status = raw.transferLinkStatus;
  if (status !== null && status !== undefined && (typeof status !== 'string' || !LINK.has(status))) return null;
  const bucket = typeof raw.cashFlowBucket === 'string' && BUCKET.has(raw.cashFlowBucket) ? raw.cashFlowBucket as FactBucket : null;
  const filings = (Array.isArray(raw.assignments) ? raw.assignments : [])
    .filter(isRecord)
    .filter((a) => typeof a.taxonomyId === 'string' && typeof a.categoryId === 'string')
    .map((a) => ({ taxonomyId: a.taxonomyId as string, categoryId: a.categoryId as string, source: typeof a.source === 'string' ? a.source : '' }));
  return {
    activityId: raw.id,
    accountId: raw.accountId,
    link: (status ?? null) as TransferLinkStatus | null,
    grouped: typeof raw.sourceGroupId === 'string' && raw.sourceGroupId.trim() !== '',
    bucket,
    filings,
  };
}

/** Linked, on Wealthfolio's word: any group at all, or a link status that is not `unlinked`. */
export function linkedOfFact(fact: CashFact): 'confirmed' | false {
  return fact.grouped || fact.link === 'linked' || fact.link === 'invalid' ? 'confirmed' : false;
}

/** What it is filed under: the spending filing when it has one, else any (as the companion's story picks). */
export function filingOfFact(fact: CashFact): FactFiling | null {
  return fact.filings.find((f) => f.taxonomyId === SPENDING_TAXONOMY_ID) ?? fact.filings[0] ?? null;
}

/** The UTC day of a stored instant (YYYY-MM-DD), or the first ten characters of what cannot be parsed. */
export function factDay(stored: string): string {
  const d = new Date(stored);
  return Number.isNaN(d.getTime()) ? stored.slice(0, 10) : d.toISOString().slice(0, 10);
}

/** The search that finds a row: its account, over the UTC day of its stored instant, both ends inclusive. */
export function cashFactQuery(row: { accountId: string; date: string }, offset = 0): {
  accountIds: string[]; startDate: string; endDate: string; offset: number; limit: number;
} {
  const day = factDay(row.date);
  return { accountIds: [row.accountId], startDate: `${day}T00:00:00.000Z`, endDate: `${day}T23:59:59.999Z`, offset, limit: CASH_FACTS_PAGE };
}
```

```ts
/**
 * src/utils/wf-facts.ts
 *
 * The add-on's reads of what Wealthfolio says about its rows (1.62.0, U3),
 * through the SDK 3.9 calls, each behind `hostCan`. A host that does not
 * offer a call (3.8), or a read that fails, gives no facts — and every
 * caller then keeps the rule it used before 3.9. Each call is made as a
 * method on `ctx.api`.
 */
import type { AddonContext, CashActivitySearchRequest } from '@wealthfolio/addon-sdk';
import { CASH_FACTS_PAGE, cashFactOf, cashFactQuery, factDay, type CashFact } from '../../shared/wf-facts';
import { hostCan } from './host-can';

/** Pages read per account and day, at most. One account's day never fills one. */
const MAX_PAGES = 3;

/** The rows' facts, by activity id — a row Wealthfolio does not return has
 *  none. Null: no facts at all (the host does not offer the call, or a read
 *  failed). */
export async function readCashFacts(
  ctx: AddonContext,
  rows: ReadonlyArray<{ id: string; accountId: string; date: string }>,
): Promise<Map<string, CashFact> | null> {
  if (rows.length === 0 || !(await hostCan(ctx, 'spending.searchCashActivities'))) return null;
  const wanted = new Set(rows.map((r) => r.id));
  const days = new Map<string, { accountId: string; date: string }>();
  for (const r of rows) days.set(`${r.accountId}\u0000${factDay(r.date)}`, r);
  const facts = new Map<string, CashFact>();
  try {
    for (const row of days.values()) {
      for (let page = 0; page < MAX_PAGES; page++) {
        const res = await ctx.api.spending.searchCashActivities(cashFactQuery(row, page * CASH_FACTS_PAGE) as CashActivitySearchRequest);
        const items = res?.items ?? [];
        for (const item of items) {
          const fact = cashFactOf(item);
          if (fact && wanted.has(fact.activityId)) facts.set(fact.activityId, fact);
        }
        if (items.length < CASH_FACTS_PAGE || (page + 1) * CASH_FACTS_PAGE >= (res?.totalCount ?? 0)) break;
      }
    }
  } catch {
    return null;
  }
  return facts;
}

/** The other half of a linked transfer. */
export interface FactPartner {
  activityId: string;
  accountId: string;
  /** YYYY-MM-DD: the UTC day of its stored instant. */
  date: string;
}

/** Wealthfolio's own pair for a row (`activities.getTransferPair`): the other
 *  half; null when the row is in no valid pair; undefined when the host does
 *  not offer the call or the read fails. */
export async function readTransferPartner(ctx: AddonContext, activityId: string): Promise<FactPartner | null | undefined> {
  if (!(await hostCan(ctx, 'activities.getTransferPair'))) return undefined;
  try {
    const pair = await ctx.api.activities.getTransferPair(activityId);
    if (!pair) return null;
    const other = pair.transferOut?.id === activityId ? pair.transferIn : pair.transferIn?.id === activityId ? pair.transferOut : null;
    if (!other || typeof other.id !== 'string' || typeof other.accountId !== 'string') return undefined;
    return { activityId: other.id, accountId: other.accountId, date: factDay(String(other.activityDate ?? '')) };
  } catch {
    return undefined;
  }
}

/** Every category's name, by id (`spending.getCategories`, all kinds —
 *  offered since Wealthfolio 3.8, so it needs no feature check); null when the
 *  host object has no such call or the read fails. */
export async function readCategoryNames(ctx: AddonContext): Promise<Map<string, string> | null> {
  if (typeof (ctx.api as { spending?: { getCategories?: unknown } } | undefined)?.spending?.getCategories !== 'function') return null;
  try {
    const categories = await ctx.api.spending.getCategories();
    return new Map((categories ?? []).map((c) => [c.categoryId, c.name]));
  } catch {
    return null;
  }
}
```

### Task 9: The one-time rewrite of a card's existing bookkeeping

Spec U4's "existing rows are rewritten once by the companion, skipping grouped rows". After each sync, before the opening-debt filing (which then has none of them left to file), the companion reads — read-only — every WITHDRAWAL on a card whose note marks it as the sync's bookkeeping (`Starting balance · `, `Balance adjustment · `, the in-transit prefix) and restates it as a TRANSFER_OUT marked external: the same amount, the stored instant verbatim (a date-only value would move the row), the same note, no symbol. It leaves alone a row in a group (an update to a linked leg is copied onto its partner), a row whose type you overrode in Wealthfolio, and a row carrying a security asset — each logged. Batches of 50; a refused or failed batch stops the run unmarked and the next sync carries on with what is still a WITHDRAWAL. One `card-transfer-out` bookkeeping entry per run that wrote anything; the secret `card_bookkeeping_rewrite` marks it done, after which it never runs again. Off on 3.9 or an unknown version, marking nothing. The audit's readers still know the rewritten rows by their notes (pinned here).

```ts
 *
 * Once, on Wealthfolio 3.10 or newer (1.62.0, U4): every bookkeeping outflow
 * the sync wrote on a credit card as a WITHDRAWAL — an opening debt, a balance
 * plug, an in-transit placeholder, known by its note — becomes a TRANSFER_OUT
 * marked external. 3.8 and 3.9 refused TRANSFER_OUT on a card (v1.55), so
 * those rows had to be WITHDRAWALs, which Wealthfolio's Spending page counts.
 * 3.10 takes one (upstream #1796) and ignores an unlinked one; the external
 * marker keeps Health from calling it an incomplete transfer. Same amount,
 * same instant, same note: the balance does not move, and the sync's readers
 * still know each row by its note.
 *
 * - Runs once: a secret records that it finished, and it never runs again.
 * - Gated: on 3.9 or an unknown version it does nothing and marks nothing.

 *   run unmarked, and the next sync finds what is still a WITHDRAWAL.

 *   linked partner), a row whose type you overrode in Wealthfolio, and a row

    ).all(STARTING_BALANCE_COMMENT_PREFIX, BALANCE_ADJUSTMENT_COMMENT_PREFIX, IN_TRANSIT_COMMENT_PREFIX) as unknown as Row[];
  } finally {
    db.close();
  }
}

const transferOut = (r: Row): ActivityWrite => ({
  id: r.id,
  accountId: r.account_id,
  activityType: 'TRANSFER_OUT',
  // The stored instant, verbatim: a date-only value would move the row.
  activityDate: r.activity_date,
  amount: Math.abs(Number(r.amount) || 0),
  fee: 0,
  currency: r.currency,
  comment: r.notes,
  metadata: EXTERNAL_BOOKKEEPING_METADATA,
});

export async function rewriteCardBookkeeping(deps: CardRewriteDeps): Promise<CardRewriteOutcome> {
  if (await deps.marker.get()) return { kind: 'done-before' };
  if (!hostTakesCardTransferOut(deps.wealthfolioVersion)) return { kind: 'older-host' };
  const rows = cardWithdrawals(deps.dbPath);
  if (rows === null) return { kind: 'no-database' };

  const left = rows.filter((r) => r.grp || r.override || (r.asset_id && !isCashSymbol(r.asset_id)));
  const todo = rows.filter((r) => !left.includes(r));
  for (const r of left) {
    const why = r.grp ? 'it is in a transfer group' : r.override ? `you typed it ${r.override} in Wealthfolio` : 'it carries an asset';
    deps.log(`Card bookkeeping rewrite: left ${r.id} as a WITHDRAWAL — ${why}.`);
  }

  const done: string[] = [];
  let stopped: string | null = null;
  for (let i = 0; i < todo.length && stopped === null; i += CARD_REWRITE_BATCH) {
    const batch = todo.slice(i, i + CARD_REWRITE_BATCH);
    try {
      const res = await deps.saveMany({ updates: batch.map(transferOut) });
      if (res.errors.length > 0) stopped = res.errors.map((e) => `${e.action}: ${e.message}`).join('; ');
      else done.push(...batch.map((r) => r.id));
    } catch (e: any) {
      stopped = String(e?.message ?? e);
    }
  }

  if (done.length > 0 && deps.record) {
    try {
      deps.record(bookkeepingChange({ step: 'card-transfer-out', activityIds: done }));
    } catch (err) {
      deps.log(`Could not record the change: ${String(err)}`);
    }
  }
  if (stopped !== null) {
    deps.log(`Card bookkeeping rewrite stopped after ${done.length} of ${todo.length} rows (${stopped}); the next sync carries on.`);
    return { kind: 'stopped', rewritten: done.length, why: stopped };
  }
  await deps.marker.set(deps.now.toISOString());
  if (done.length > 0) deps.log(`Restated ${done.length} card bookkeeping row(s) as transfers out, which Wealthfolio leaves out of spending.`);
  return { kind: 'done', rewritten: done.length, left: left.length };
}
```
